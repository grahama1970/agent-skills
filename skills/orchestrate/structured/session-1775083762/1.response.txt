"""
Chaos Engineering and Novel Exploitation logic for the hack skill.

Uses high-reasoning models to brainstorm insane
exploit ideas and generate novel payloads, including assembly.
"""
from __future__ import annotations

import json
from typing import List, Dict, Any, Optional

import httpx
from rich.console import Console

from hack.config import DEFAULT_MODEL
from loguru import logger

console = Console()

SCILLM_URL = "http://localhost:4001"
SCILLM_CHAT_ENDPOINT = f"{SCILLM_URL}/chat"


def _map_model(model: str) -> str:
    """Map legacy/default model names to scillm-supported model names."""
    if model == "gpt-5.2-codex":
        return "gpt-5.3-codex"
    return model


def _scillm_chat(prompt: str, model: str) -> str:
    """
    Send a prompt to scillm chat endpoint and return response text.

    Raises exceptions on network/protocol failures so callers can fallback.
    """
    mapped_model = _map_model(model)
    payload = {
        "model": mapped_model,
        "messages": [{"role": "user", "content": prompt}],
    }

    resp = httpx.post(SCILLM_CHAT_ENDPOINT, json=payload, timeout=60.0)
    resp.raise_for_status()

    data = resp.json()
    if isinstance(data, dict):
        # OpenAI-like shape
        choices = data.get("choices")
        if isinstance(choices, list) and choices:
            msg = choices[0].get("message", {})
            content = msg.get("content")
            if isinstance(content, str):
                return content

        # Common direct response keys
        for key in ("content", "response", "text", "output"):
            val = data.get(key)
            if isinstance(val, str):
                return val

    # Last resort: stringify body for debugging/fallback parsing
    return json.dumps(data)


def brainstorm_chaos_exploits(
    target: str,
    findings: List[Dict[str, Any]] = None,
    failure_context: Optional[str] = None,
    model: str = DEFAULT_MODEL
) -> List[Dict[str, Any]]:
    """
    Use scillm to brainstorm novel/insane exploit ideas for a target.
    Includes failure analysis for iterative feedback loops.
    """
    prompt = f"""
    Target: {target}
    Previous Findings: {json.dumps(findings) if findings else 'None'}
    
    FAILURE CONTEXT FROM PREVIOUS ATTEMPT:
    {failure_context if failure_context else 'No previous attempt or context available.'}
    
    Task: Brainstorm 3 'insane' or novel exploit ideas. 
    If a failure context is provided, analyze the error (stdout/stderr) 
    and suggest a way to bypass or fix the specific failure.
    Think outside the box. Consider:
    - Timing side-channels
    - Exotic encodings
    - Assembly-level manipulation
    - Race conditions in unlikely places
    - Novel protocol abuse
    
    Return the ideas as a JSON list of objects with:
    - title: Brief name of the idea
    - description: How it works
    - insane_factor: 1-10 rating of how 'novel' it is
    - payload_hint: A conceptual payload or snippet
    """

    try:
        stdout = _scillm_chat(prompt, model)
        start = stdout.find("[")
        end = stdout.rfind("]") + 1
        if start != -1 and end != 0:
            parsed = json.loads(stdout[start:end])
            if isinstance(parsed, list):
                return parsed
    except Exception as e:
        console.print(f"[dim]Chaos brainstorming failed: {e}[/dim]")

    return baseline_chaos_ideas(target)


def baseline_chaos_ideas(target: str) -> List[Dict[str, Any]]:
    """Fallback list of novel exploit ideas."""
    return [
        {
            "title": "Unicode Normalization Bypass",
            "description": "Using exotic Unicode characters that normalize to restricted ones (e.g., self-XSS via Turkish dotless I).",
            "insane_factor": 6,
            "payload_hint": "ı -> i bypass"
        },
        {
            "title": "Assembly-level Shellcode Injection",
            "description": "Crafting handcrafted assembly payloads to evade pattern-matching WAFs.",
            "insane_factor": 8,
            "payload_hint": "section .text; global _start; _start: ..."
        }
    ]


def generate_assembly_payload(goal: str, model: str = DEFAULT_MODEL) -> str:
    """Use scillm to generate a specific assembly payload."""
    prompt = f"Write a minimal Linux x64 assembly (NASM) snippet to achieve: {goal}. Provide ONLY the code."

    try:
        stdout = _scillm_chat(prompt, model)
        return stdout.strip()
    except Exception as e:
        logger.debug("value extraction failed: {}", e)
    return "; Error: Payload generation failed."


__all__ = ["brainstorm_chaos_exploits", "generate_assembly_payload"]
