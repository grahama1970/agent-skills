function FrameworkDonut({ item }: { item: FrameworkRing }) {
  const size = 94
  const outerR = 34
  const innerR = 24
  const c = size / 2

  const arcGen = useMemo(() => arc<{ startAngle: number; endAngle: number }>().innerRadius(innerR).outerRadius(outerR), [innerR, outerR])
  const slices = useMemo(
    () =>
      pie<number>()
        .value((d) => d)
        .sort(null)([item.pass, item.partial, item.fail]),
    [item.pass, item.partial, item.fail],
  )
  const fwColor = EMBRY.fw?.[item.framework] || EMBRY.green

  return (
    <button
      type="button"
      title={`${item.framework} framework coverage`}
      data-qid={`posture:donut:${item.framework}`}
      data-qs-action={`POSTURE_SELECT_FRAMEWORK_${item.framework}`}
      style={{ border: `1px solid ${EMBRY.bg}`, borderRadius: 10, background: EMBRY.bg, padding: 8, cursor: 'pointer', outlineOffset: 2 }}
    >
      <div style={{ ...label, marginBottom: 6 }}>{item.framework}</div>
      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} aria-hidden>
        <g transform={`translate(${c},${c})`}>
          {slices.map((s, i) => {
            const fill = i === 0 ? fwColor : i === 1 ? EMBRY.amber : EMBRY.red
            return <path key={`${item.framework}-${i}`} d={arcGen(s) || ''} fill={fill} />
          })}
        </g>
        <text x="50%" y="50%" textAnchor="middle" dominantBaseline="middle" style={{ ...heading, fontSize: 14, fill: EMBRY.green }}>
          {item.pct}%
        </text>
      </svg>
      <div style={{ ...label }}>P:{item.pass} A:{item.partial} F:{item.fail}</div>
    </button>
  )
}
