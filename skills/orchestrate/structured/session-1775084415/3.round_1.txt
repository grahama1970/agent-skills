from __future__ import annotations

import json
import subprocess
import tempfile
import uuid
from pathlib import Path
from typing import Any

import httpx
from loguru import logger
from pydantic import BaseModel, Field

from common.file_bundler import bundle_for_review


class ExploitResult(BaseModel):
    success: bool = False
    rounds: int = 0
    exploit_code: str = ""
    score: float = Field(default=0.0)


def _recall_memory(query: str, limit: int = 5) -> str:
    """Best-effort memory recall for prior exploit context."""
    try:
        # Optional local memory endpoint; fail open if unavailable.
        with httpx.Client(timeout=3.0) as client:
            resp = client.post(
                "http://127.0.0.1:8000/recall",
                json={"query": query, "limit": limit},
            )
            if resp.status_code != 200:
                return ""
            data: Any = resp.json()
            if isinstance(data, dict):
                items = data.get("items") or data.get("results") or []
            elif isinstance(data, list):
                items = data
            else:
                items = []
            lines: list[str] = []
            for item in items[:limit]:
                if isinstance(item, dict):
                    txt = item.get("text") or item.get("content") or str(item)
                else:
                    txt = str(item)
                if txt:
                    lines.append(f"- {txt}")
            return "\n".join(lines)
    except Exception as exc:
        logger.debug(f"Memory recall unavailable: {exc}")
        return ""


def write_exploit(
    target_path: str,
    finding: str,
    prior_context: str,
    docker_target: str,
    max_rounds: int = 5,
) -> ExploitResult:
    """Generate an exploit with /code-runner and return structured result."""
    skill_dir = Path(__file__).resolve().parent
    code_runner_dir = skill_dir.parent / "code-runner"
    run_sh = code_runner_dir / "run.sh"

    target = Path(target_path).resolve()
    cwd = str(skill_dir)

    target_files = [str(target)] if target.exists() else []
    bundled = {}
    if target_files:
        try:
            bundled = bundle_for_review(target_files, cwd=str(Path.cwd()))
        except Exception as exc:
            logger.warning(f"Failed to bundle target context: {exc}")

    memory = _recall_memory(f"exploit {target.name} {finding}")
    memory_block = memory if memory else "(no prior exploit memories found)"

    task_id = f"exploit-{uuid.uuid4().hex[:10]}"
    exploit_rel = f"exploits/{task_id}.py"
    exploit_abs = skill_dir / exploit_rel

    read_context = []
    for p, content in bundled.items():
        read_context.append({"path": p, "content": content})

    prompt = f"""Write a Python exploit script to validate this finding.

Target path: {target}
Finding:
{finding}

Prior context:
{prior_context}

Memory recall (prior exploits):
{memory_block}

Requirements:
- Write exploit to: {exploit_rel}
- Keep code focused, minimal dependencies, and runnable with python3.
- Script must exit 0 on successful exploitation/validation.
- Prefer deterministic checks and clear stdout.

Docker target/container: {docker_target}
"""

    dod_cmd = f"docker exec {docker_target} python3 {exploit_rel}"
    spec: dict[str, Any] = {
        "task_id": task_id,
        "cwd": cwd,
        "backend": "codex",
        "prompt": prompt,
        "allowlist": [str(skill_dir / "exploits")],
        "read_context": read_context,
        "definition_of_done": {
            "commands": [dod_cmd],
            "assertions": ["exit_code == 0"],
        },
    }

    try:
        (skill_dir / "exploits").mkdir(parents=True, exist_ok=True)
        with tempfile.TemporaryDirectory(prefix="exploit-writer-") as td:
            spec_file = Path(td) / "spec.json"
            spec_file.write_text(json.dumps(spec, indent=2), encoding="utf-8")

            cmd = [str(run_sh), "run", str(spec_file), "--max-rounds", str(max_rounds)]
            logger.info(f"Running code-runner for exploit generation: {' '.join(cmd)}")
            proc = subprocess.run(
                cmd,
                cwd=str(code_runner_dir),
                text=True,
                capture_output=True,
                check=False,
            )
            if proc.returncode != 0:
                logger.error(f"code-runner failed: {proc.stderr.strip()}")
                return ExploitResult(success=False, rounds=0, exploit_code="", score=0.0)

            result_file = code_runner_dir / "result.json"
            if not result_file.exists():
                logger.error("code-runner result.json not found")
                return ExploitResult(success=False, rounds=0, exploit_code="", score=0.0)

            result = json.loads(result_file.read_text(encoding="utf-8"))
            success = bool(result.get("success", False))
            rounds = int(result.get("rounds") or result.get("round_count") or 0)

            score = 0.0
            if isinstance(result.get("score"), (int, float)):
                score = float(result["score"])
            elif success:
                score = 1.0

            exploit_code = ""
            if exploit_abs.exists():
                exploit_code = exploit_abs.read_text(encoding="utf-8", errors="replace")
            else:
                files = result.get("files_written") or result.get("artifacts") or []
                for fp in files:
                    p = Path(fp)
                    if p.name == exploit_abs.name and p.exists():
                        exploit_code = p.read_text(encoding="utf-8", errors="replace")
                        break

            return ExploitResult(
                success=success,
                rounds=rounds,
                exploit_code=exploit_code,
                score=score,
            )
    except Exception as exc:
        logger.exception(f"write_exploit failed: {exc}")
        return ExploitResult(success=False, rounds=0, exploit_code="", score=0.0)
