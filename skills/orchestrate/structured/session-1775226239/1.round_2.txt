/**
 * UX Lab API Server
 * 
 *
 * Thin proxy layer — all data flows through the memory daemon Unix socket.
 * No bespoke reimplementations of skill logic. No hardcoded file paths.
 *
 * Routes:
 *   GET  /api/health           — server health check
 *   ALL  /api/memory/*         — proxy to memory daemon (recall, learn, list, etc.)
 *   POST /api/scillm           — proxy to scillm LLM gateway
 *   GET  /api/models           — discover available LLM models (grouped)
 *   GET  /api/projects/:id/models — ModelPicker-format model registry
 */

function toWorksheetRecord(value: unknown): Record<string, WorksheetConfig> {
  if (!value || typeof value !== 'object' || Array.isArray(value)) return {}
  const out: Record<string, WorksheetConfig> = {}
  for (const [key, worksheet] of Object.entries(value)) {
    if (worksheet && typeof worksheet === 'object' && !Array.isArray(worksheet)) {
      out[key] = worksheet as WorksheetConfig
    }
  }
  return out
}

function parseWorksheetsYaml(content: string): Record<string, WorksheetConfig> {
  const parsed = yamlLoad(content)
  if (parsed && typeof parsed === 'object' && !Array.isArray(parsed) && 'worksheets' in parsed) {
    const nested = (parsed as { worksheets?: unknown }).worksheets
    return toWorksheetRecord(nested)
  }
  return toWorksheetRecord(parsed)
}

app.get('/api/posture/frameworks', async (_req, res) => {
  const normalizeListItems = (payload: any): any[] => (Array.isArray(payload?.items) ? payload.items : (Array.isArray(payload) ? payload : []))
  const postToMemoryList = (payload: Record<string, unknown>): Promise<any> =>
    new Promise((resolve, reject) => {
      const body = JSON.stringify(payload)
      const req = httpRequest(
        {
          socketPath: MEMORY_SOCKET,
          path: '/list',
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Content-Length': Buffer.byteLength(body),
          },
        },
        (resp) => {
          let data = ''
          resp.on('data', (chunk) => { data += String(chunk) })
          resp.on('end', () => {
            if ((resp.statusCode ?? 500) >= 400) {
              reject(new Error(`Memory daemon returned HTTP ${resp.statusCode}`))
              return
            }
            try {
              if (!data || !data.trim()) {
                resolve({ items: [] })
                return
              }
              resolve(JSON.parse(data))
            } catch (err) {
              reject(new Error('Invalid JSON from memory daemon'))
            }
          })
        }
      )
      req.on('error', (err: any) => reject(err))
      req.write(body)
      req.end()
    })

  try {
    const [controlsResp, qraResp] = await Promise.all([
      postToMemoryList({
        collection: 'sparta_controls',
        return_fields: ['control_id', 'source_framework', 'nrs_score', 'weaknesses'],
        limit: 0,
      }),
      postToMemoryList({
        collection: 'sparta_qra',
        return_fields: ['control_id'],
        group_by: 'control_id',
        limit: 0,
      }),
    ])

    const controls: any[] = normalizeListItems(controlsResp)
    const qraItems: any[] = normalizeListItems(qraResp)
    const qraControlIds = new Set<string>(qraItems.map((r) => String(r?.control_id ?? '')).filter(Boolean))

    const frameworkMap = new Map<string, { name: string; total: number; withQRAs: number; pct: number }>()
    for (const c of controls) {
      const name = String(c?.source_framework ?? 'Unknown')
      const controlId = String(c?.control_id ?? '')
      if (!frameworkMap.has(name)) frameworkMap.set(name, { name, total: 0, withQRAs: 0, pct: 0 })
      const entry = frameworkMap.get(name)!
      entry.total += 1
      if (controlId && qraControlIds.has(controlId)) entry.withQRAs += 1
    }

    const frameworks = Array.from(frameworkMap.values()).map((f) => ({
      ...f,
      pct: f.total > 0 ? Math.round((f.withQRAs / f.total) * 100) : 0,
    }))

    const totalControls = frameworks.reduce((acc, f) => acc + f.total, 0)
    const weightedPctSum = frameworks.reduce((acc, f) => acc + (f.pct * f.total), 0)
    const overallScore = totalControls > 0 ? Math.round(weightedPctSum / totalControls) : 0

    res.json({ overallScore, delta: 0, frameworks })
  } catch {
    res.status(502).json({ error: 'Memory daemon unavailable' })
  }
})
