          <div style={{ display: 'grid', gap: 8 }}>

                key={a.control_id}

            </thead>
            <tbody>
              {RISKS.map((r) => (
                <tr key={r.control_id}>
                  <td colSpan={4} style={{ padding: 0 }}>
                    <button
                      type="button"

        </div>
      </section>

      <aside style={{ ...card, background: EMBRY.bgCard, padding: 12 }}>
        <div style={heading}>Family Flyout</div>
        <div style={label}>{selectedFamily ? `Selected family: ${selectedFamily}` : 'Select a family bar to view detail.'}</div>

  )
