    { control_id: 'PL-2', control: 'System Security Plan', framework: 'RMF', family: 'PL', risk: 74, trend: 'up' }
  ]

  const postureScore = 79
  const scoreDelta = +3

  const donutArcs = useMemo(() => {
    const pieGen = d3.pie<number>().sort(null)
    const arcGen = d3.arc<d3.PieArcDatum<number>>().innerRadius(22).outerRadius(32)
    return frameworkDonuts.reduce<Record<string, string[]>>((acc, item) => {
      const arcs = pieGen([item.pass, item.partial, item.fail]).map((a) => arcGen(a) || '')
      acc[item.framework] = arcs
      return acc
    }, {})
  }, [])

  const selectedFamilyData = familyData.find((f) => f.family === selectedFamily) || null

  const getSeverityColor = (severity: AlertItem['severity']) => {
    if (severity === 'high') return EMBRY.red
    if (severity === 'medium') return EMBRY.amber
    return EMBRY.green
  }

  return (
    <div style={{ ...card, background: EMBRY.bg, minHeight: '100%', padding: 16, display: 'grid', gap: 12 }}>
      <div style={{ display: 'grid', gridTemplateColumns: '1.2fr 2fr', gap: 12 }}>
        <section style={{ ...card, background: EMBRY.bgCard, padding: 16 }}>
          <div style={{ ...label }}>Overall Posture</div>
          <div style={{ ...heading, fontSize: 56, lineHeight: 1.05, color: EMBRY.green, textShadow: `0 0 12px ${EMBRY.green}` }}>{postureScore}%</div>
          <div style={{ display: 'inline-flex', alignItems: 'center', gap: 8, ...fwBadge }}>
            <span aria-hidden="true" style={{ ...glowDot, background: scoreDelta >= 0 ? EMBRY.green : EMBRY.red }} />
            <span style={{ ...label }}>{scoreDelta >= 0 ? `+${scoreDelta}%` : `${scoreDelta}%`} vs last scan</span>
          </div>
        </section>
        <section style={{ ...card, background: EMBRY.bgCard, padding: 16 }}>
          <div style={{ ...heading, fontSize: 16, marginBottom: 10 }}>Framework Coverage</div>
          <div style={{ display: 'grid', gridTemplateColumns: 'repeat(8, minmax(88px, 1fr))', gap: 10 }}>
            {frameworkDonuts.map((item) => {
              const fwColor = EMBRY.fw[item.framework] || EMBRY.green
              const arcs = donutArcs[item.framework] || []
              return (
                <button
                  key={item.framework}
                  type="button"
                  title={`${item.framework} coverage ${item.score}%`}
                  data-qid={`posture:donut:${item.framework}`}
                  data-qs-action={`POSTURE_SELECT_FRAMEWORK_${item.framework}`}
                  style={{ ...card, background: EMBRY.bgCard, border: `1px solid ${fwColor}`, padding: 8, cursor: 'pointer' }}
                >
                  <svg width="72" height="72" viewBox="0 0 80 80" role="img" aria-label={`${item.framework} donut`}>
                    <g transform="translate(40,40)">
                      <circle r="32" fill="none" stroke={EMBRY.bg} strokeWidth="8" />
                      {arcs.map((d, idx) => (
                        <path key={`${item.framework}-${idx}`} d={d} fill={idx === 0 ? fwColor : idx === 1 ? EMBRY.amber : EMBRY.red} />
                      ))}
                    </g>
                    <text x="40" y="44" textAnchor="middle" style={{ ...label, fill: EMBRY.text }}>{item.score}%</text>
                  </svg>
                  <div style={{ ...label, marginTop: 4 }}>{item.framework}</div>
                </button>
              )
            })}
          </div>
        </section>
      </div>

      <div style={{ display: 'grid', gridTemplateColumns: '2fr 1fr', gap: 12 }}>
        <section style={{ ...card, background: EMBRY.bgCard, padding: 16 }}>
          <div style={{ ...heading, fontSize: 16, marginBottom: 10 }}>NIST 800-53 Families</div>
          <div style={{ display: 'grid', gap: 8 }}>
            {familyData.map((fam) => {
              const passPct = (fam.pass / fam.total) * 100
              const partPct = (fam.partial / fam.total) * 100
              const failPct = (fam.fail / fam.total) * 100
              return (
                <button
                  key={fam.family}
                  type="button"
                  title={`Open family ${fam.family} details`}
                  data-qid={`posture:family:${fam.family}`}
                  data-qs-action={`POSTURE_SELECT_FAMILY_${fam.family}`}
                  onClick={() => setSelectedFamily(fam.family)}
                  style={{ display: 'grid', gridTemplateColumns: '52px 1fr 52px', alignItems: 'center', gap: 8, background: 'transparent', border: 0, cursor: 'pointer' }}
                >
                  <span style={{ ...label }}>{fam.family}</span>
                  <span style={{ display: 'flex', height: 14, borderRadius: 999, overflow: 'hidden', border: `1px solid ${EMBRY.textDim}` }}>
                    <span style={{ width: `${passPct}%`, background: EMBRY.green }} />
                    <span style={{ width: `${partPct}%`, background: EMBRY.amber }} />
                    <span style={{ width: `${failPct}%`, background: EMBRY.red }} />
                  </span>
                  <span style={{ ...label }}>{Math.round((fam.pass / fam.total) * 100)}%</span>
                </button>
              )
            })}
          </div>
          <div style={{ ...label, marginTop: 8 }}>Legend: Pass / Partial / Fail</div>
        </section>
        <section style={{ ...card, background: EMBRY.bgCard, padding: 16 }}>
          <div style={{ ...heading, fontSize: 16, marginBottom: 10 }}>Gap Analysis</div>
          <div style={{ display: 'grid', gap: 8 }}>
            {gapData.map((g) => (
              <button
                key={g.type}
                type="button"
                title={`${g.label}: ${g.detail}`}
                data-qid={`posture:gap:${g.type}`}
                data-qs-action={`POSTURE_VIEW_GAP_${g.type.toUpperCase()}`}
                style={{ ...card, background: EMBRY.bg, border: `1px solid ${EMBRY.textDim}`, padding: 8, textAlign: 'left', cursor: 'pointer' }}
              >
                <div style={{ ...label }}>{g.label}</div>
                <div style={{ ...heading, fontSize: 20 }}>{g.value}</div>
                <div style={{ ...label }}>{g.detail}</div>
              </button>
            ))}
          </div>
        </section>
      </div>

      <div style={{ display: 'grid', gridTemplateColumns: '1.1fr 1.9fr', gap: 12 }}>
        <section style={{ ...card, background: EMBRY.bgCard, padding: 16 }}>
          <div style={{ ...heading, fontSize: 16, marginBottom: 10 }}>Drift Alerts</div>
          <div style={{ display: 'grid', gap: 8 }}>
            {alerts.length === 0 ? (
              <div style={{ ...label }}>No current drift alerts.</div>
            ) : (
              alerts.map((a) => (
                <button
                  key={a.control_id}
                  type="button"
                  title={`${a.control_id} ${a.severity} drift alert`}
                  data-qid={`posture:alert:${a.control_id}`}
                  data-qs-action={`POSTURE_VIEW_ALERT_${a.control_id}`}
                  style={{ display: 'grid', gridTemplateColumns: '16px 1fr', gap: 8, alignItems: 'start', background: 'transparent', border: 0, textAlign: 'left', cursor: 'pointer' }}
                >
                  <span aria-hidden="true" style={{ ...glowDot, background: getSeverityColor(a.severity), marginTop: 3 }} />
                  <span>
                    <span style={{ ...label }}>{a.control_id} · {a.framework} · {a.severity.toUpperCase()}</span>
                    <span style={{ display: 'block', ...label }}>{a.message}</span>
                  </span>
                </button>
              ))
            )}
          </div>
        </section>
        <section style={{ ...card, background: EMBRY.bgCard, padding: 16 }}>
          <div style={{ ...heading, fontSize: 16, marginBottom: 10 }}>Top 10 Risk Controls</div>
          <div role="table" aria-label="Top risk controls table" style={{ display: 'grid', gap: 4 }}>
            <div role="row" style={{ display: 'grid', gridTemplateColumns: '90px 1.2fr 72px 80px 72px', gap: 8, ...label }}>
              <span>Control ID</span><span>Control</span><span>Framework</span><span>Family</span><span>Risk</span>
            </div>
            {risks.map((r) => (
              <button
                key={r.control_id}
                type="button"
                title={`Open control ${r.control_id} detail`}
                data-qid={`posture:risk:${r.control_id}`}
                data-qs-action={`POSTURE_VIEW_RISK_${r.control_id}`}
                onClick={() => onNavigateToControl?.(r.control_id)}
                style={{ display: 'grid', gridTemplateColumns: '90px 1.2fr 72px 80px 72px', gap: 8, alignItems: 'center', textAlign: 'left', background: EMBRY.bg, border: `1px solid ${EMBRY.textDim}`, padding: 8, cursor: 'pointer' }}
              >
                <span style={{ ...label }}>{r.control_id}</span>
                <span style={{ ...label }}>{r.control}</span>
                <span style={{ ...label }}>{r.framework}</span>
                <span style={{ ...label }}>{r.family}</span>
                <span style={{ ...label }}>{r.risk} {r.trend === 'up' ? '↑' : r.trend === 'down' ? '↓' : '→'}</span>
              </button>
            ))}
          </div>
        </section>
      </div>

      {selectedFamilyData ? (
        <aside style={{ ...card, background: EMBRY.bgCard, padding: 12 }}>
          <div style={{ ...heading, fontSize: 14 }}>{selectedFamilyData.family} Family Details</div>
          <div style={{ ...label }}>Pass: {selectedFamilyData.pass} · Partial: {selectedFamilyData.partial} · Fail: {selectedFamilyData.fail}</div>
          <div style={{ ...label }}>Total: {selectedFamilyData.total} controls</div>
        </aside>
      ) : (
        <aside style={{ ...card, background: EMBRY.bgCard, padding: 12 }}>
          <div style={{ ...label }}>Select a family bar to view details.</div>
        </aside>
      )}
    </div>
  )
}
