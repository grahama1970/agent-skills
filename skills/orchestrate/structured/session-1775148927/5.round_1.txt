from src.calculator import add, subtract, divide

    """Test subtracting floating point numbers."""
    result = subtract(5.5, 2.5)
    assert result == 3.0


def test_divide_returns_float():
    """Test division returns float even with integers."""
    result = divide(7, 2)
    assert result == 3.5

    """Test formatting negative currency amounts."""
    result = format_currency(-100)
    assert result == "-$100.00"

    """Test formatting a list with single item."""
    result = format_list(["apple"])
